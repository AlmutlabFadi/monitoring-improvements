from flask import Blueprint, request, jsonify, send_file
from src.models.device import db, Device, AudioFile, Screenshot, ActivityLog
from datetime import datetime, timedelta
import os
import json

devices_bp = Blueprint('devices', __name__)

@devices_bp.route('/devices', methods=['GET'])
def get_devices():
    """Get all devices"""
    try:
        devices = Device.query.all()
        return jsonify({
            'success': True,
            'devices': [device.to_dict() for device in devices]
        })
    except Exception as e:
        return jsonify({'success': False, 'error': str(e)}), 500

@devices_bp.route('/devices', methods=['POST'])
def register_device():
    """Register a new device"""
    try:
        data = request.get_json()
        
        # Check if device already exists
        existing_device = Device.query.get(data['id'])
        if existing_device:
            return jsonify({'success': False, 'error': 'Device already exists'}), 400
        
        device = Device(
            id=data['id'],
            name=data['name'],
            device_type=data.get('device_type', 'android'),
            status='connected',
            battery_level=data.get('battery_level', 0),
            location=data.get('location', '')
        )
        
        db.session.add(device)
        db.session.commit()
        
        return jsonify({
            'success': True,
            'device': device.to_dict()
        }), 201
    except Exception as e:
        return jsonify({'success': False, 'error': str(e)}), 500

@devices_bp.route('/devices/<device_id>/services', methods=['PUT'])
def update_device_services(device_id):
    """Update device services status"""
    try:
        device = Device.query.get_or_404(device_id)
        data = request.get_json()
        
        if 'audio_recording' in data:
            device.audio_recording = data['audio_recording']
        if 'screenshot_capture' in data:
            device.screenshot_capture = data['screenshot_capture']
        if 'activity_monitoring' in data:
            device.activity_monitoring = data['activity_monitoring']
        
        device.last_activity = datetime.utcnow()
        db.session.commit()
        
        return jsonify({
            'success': True,
            'device': device.to_dict()
        })
    except Exception as e:
        return jsonify({'success': False, 'error': str(e)}), 500

@devices_bp.route('/devices/<device_id>/audio', methods=['POST'])
def upload_audio_file(device_id):
    """Upload audio file for a device"""
    try:
        device = Device.query.get_or_404(device_id)
        
        if 'file' not in request.files:
            return jsonify({'success': False, 'error': 'No file provided'}), 400
        
        file = request.files['file']
        if file.filename == '':
            return jsonify({'success': False, 'error': 'No file selected'}), 400
        
        # Create uploads directory if it doesn't exist
        upload_dir = os.path.join(os.path.dirname(__file__), '..', 'uploads', 'audio', device_id)
        os.makedirs(upload_dir, exist_ok=True)
        
        # Save file
        filename = f"{datetime.utcnow().strftime('%Y%m%d_%H%M%S')}_{file.filename}"
        file_path = os.path.join(upload_dir, filename)
        file.save(file_path)
        
        # Get file size
        file_size = os.path.getsize(file_path)
        
        # Save to database
        audio_file = AudioFile(
            device_id=device_id,
            filename=filename,
            file_path=file_path,
            file_size=file_size,
            duration=request.form.get('duration', type=int)
        )
        
        db.session.add(audio_file)
        db.session.commit()
        
        return jsonify({
            'success': True,
            'audio_file': audio_file.to_dict()
        }), 201
    except Exception as e:
        return jsonify({'success': False, 'error': str(e)}), 500

# Statistics Routes
@devices_bp.route('/stats', methods=['GET'])
def get_statistics():
    """Get overall statistics"""
    try:
        total_devices = Device.query.count()
        connected_devices = Device.query.filter_by(status='connected').count()
        total_audio_files = AudioFile.query.count()
        total_screenshots = Screenshot.query.count()
        
        # Calculate storage usage (in bytes)
        audio_storage = db.session.query(db.func.sum(AudioFile.file_size)).scalar() or 0
        screenshot_storage = db.session.query(db.func.sum(Screenshot.file_size)).scalar() or 0
        total_storage = audio_storage + screenshot_storage
        
        return jsonify({
            'success': True,
            'stats': {
                'total_devices': total_devices,
                'connected_devices': connected_devices,
                'total_audio_files': total_audio_files,
                'total_screenshots': total_screenshots,
                'storage_used_bytes': total_storage,
                'storage_used_mb': round(total_storage / (1024 * 1024), 2)
            }
        })
    except Exception as e:
        return jsonify({'success': False, 'error': str(e)}), 500
