from flask_sqlalchemy import SQLAlchemy
from datetime import datetime
import json

db = SQLAlchemy()

class Device(db.Model):
    __tablename__ = 'devices'
    
    id = db.Column(db.String(50), primary_key=True)
    name = db.Column(db.String(100), nullable=False)
    device_type = db.Column(db.String(50), nullable=False)  # android, ios
    status = db.Column(db.String(20), default='offline')  # connected, idle, offline
    battery_level = db.Column(db.Integer, default=0)
    location = db.Column(db.String(200))
    last_activity = db.Column(db.DateTime, default=datetime.utcnow)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    
    # Services status
    audio_recording = db.Column(db.Boolean, default=False)
    screenshot_capture = db.Column(db.Boolean, default=False)
    activity_monitoring = db.Column(db.Boolean, default=False)
    
    # Relationships
    audio_files = db.relationship('AudioFile', backref='device', lazy=True, cascade='all, delete-orphan')
    screenshots = db.relationship('Screenshot', backref='device', lazy=True, cascade='all, delete-orphan')
    activities = db.relationship('ActivityLog', backref='device', lazy=True, cascade='all, delete-orphan')
    
    def to_dict(self):
        return {
            'id': self.id,
            'name': self.name,
            'device_type': self.device_type,
            'status': self.status,
            'battery_level': self.battery_level,
            'location': self.location,
            'last_activity': self.last_activity.isoformat() if self.last_activity else None,
            'created_at': self.created_at.isoformat() if self.created_at else None,
            'services': {
                'audio_recording': self.audio_recording,
                'screenshot_capture': self.screenshot_capture,
                'activity_monitoring': self.activity_monitoring
            }
        }
    
    def update_status(self, status, battery_level=None, location=None):
        self.status = status
        self.last_activity = datetime.utcnow()
        if battery_level is not None:
            self.battery_level = battery_level
        if location:
            self.location = location
        db.session.commit()

class AudioFile(db.Model):
    __tablename__ = 'audio_files'
    
    id = db.Column(db.Integer, primary_key=True)
    device_id = db.Column(db.String(50), db.ForeignKey('devices.id'), nullable=False)
    filename = db.Column(db.String(200), nullable=False)
    file_path = db.Column(db.String(500), nullable=False)
    duration = db.Column(db.Integer)  # in seconds
    file_size = db.Column(db.Integer)  # in bytes
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    
    def to_dict(self):
        return {
            'id': self.id,
            'device_id': self.device_id,
            'filename': self.filename,
            'duration': self.duration,
            'file_size': self.file_size,
            'created_at': self.created_at.isoformat() if self.created_at else None
        }

class Screenshot(db.Model):
    __tablename__ = 'screenshots'
    
    id = db.Column(db.Integer, primary_key=True)
    device_id = db.Column(db.String(50), db.ForeignKey('devices.id'), nullable=False)
    filename = db.Column(db.String(200), nullable=False)
    file_path = db.Column(db.String(500), nullable=False)
    file_size = db.Column(db.Integer)  # in bytes
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    
    def to_dict(self):
        return {
            'id': self.id,
            'device_id': self.device_id,
            'filename': self.filename,
            'file_size': self.file_size,
            'created_at': self.created_at.isoformat() if self.created_at else None
        }

class ActivityLog(db.Model):
    __tablename__ = 'activity_logs'
    
    id = db.Column(db.Integer, primary_key=True)
    device_id = db.Column(db.String(50), db.ForeignKey('devices.id'), nullable=False)
    activity_type = db.Column(db.String(50), nullable=False)  # text_input, app_usage, etc.
    content = db.Column(db.Text)
    meta_data = db.Column(db.Text)  # JSON string for additional data
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    
    def to_dict(self):
        return {
            'id': self.id,
            'device_id': self.device_id,
            'activity_type': self.activity_type,
            'content': self.content,
            'metadata': json.loads(self.meta_data) if self.meta_data else None,
            'created_at': self.created_at.isoformat() if self.created_at else None
        }
