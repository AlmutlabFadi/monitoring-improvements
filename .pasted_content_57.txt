رائع! سأبدأ بتصميم هيكل منظم لتطبيق الكاميرا المتقدم، باستخدام كلاس CameraController لإدارة كل عمليات الكاميرا (المعاينة، التسجيل، التبديل بين الكاميرات) مع دعم الأذونات وإدارة الأداء.

1. إنشاء كلاس CameraController كامل مع التعليقات

import android.Manifest
import android.content.Context
import android.content.pm.PackageManager
import android.hardware.camera2.*
import android.media.MediaRecorder
import android.os.Environment
import android.util.Log
import android.view.Surface
import androidx.core.app.ActivityCompat
import java.io.File
import java.text.SimpleDateFormat
import java.util.*

class CameraController(
    private val context: Context,
    private val cameraManager: CameraManager,
    private val previewSurface: Surface
) {

    private var cameraDevice: CameraDevice? = null
    private var captureSession: CameraCaptureSession? = null
    private var mediaRecorder: MediaRecorder? = null

    var isFrontCamera = true
        private set
    private lateinit var cameraId: String

    var isRecording = false
        private set

    private val TAG = "CameraController"

    // فتح الكاميرا (أمامية أو خلفية حسب isFrontCamera)
    fun openCamera(onSuccess: () -> Unit, onError: (String) -> Unit) {
        try {
            cameraId = cameraManager.cameraIdList.first {
                val characteristics = cameraManager.getCameraCharacteristics(it)
                val facing = characteristics.get(CameraCharacteristics.LENS_FACING)
                if (isFrontCamera) facing == CameraCharacteristics.LENS_FACING_FRONT
                else facing == CameraCharacteristics.LENS_FACING_BACK
            }

            if (ActivityCompat.checkSelfPermission(context, Manifest.permission.CAMERA) != PackageManager.PERMISSION_GRANTED) {
                onError("Camera permission not granted")
                return
            }

            cameraManager.openCamera(cameraId, object : CameraDevice.StateCallback() {
                override fun onOpened(camera: CameraDevice) {
                    cameraDevice = camera
                    onSuccess()
                }

                override fun onDisconnected(camera: CameraDevice) {
                    camera.close()
                    cameraDevice = null
                }

                override fun onError(camera: CameraDevice, error: Int) {
                    onError("Camera error: $error")
                    camera.close()
                    cameraDevice = null
                }
            }, null)

        } catch (e: CameraAccessException) {
            onError("CameraAccessException: ${e.message}")
        }
    }

    // بدء معاينة الكاميرا
    fun startPreview(onPreviewStarted: () -> Unit, onPreviewFailed: () -> Unit) {
        val camera = cameraDevice ?: run {
            Log.e(TAG, "startPreview: CameraDevice is null")
            onPreviewFailed()
            return
        }

        try {
            val previewRequestBuilder = camera.createCaptureRequest(CameraDevice.TEMPLATE_PREVIEW)
            previewRequestBuilder.addTarget(previewSurface)

            camera.createCaptureSession(listOf(previewSurface), object : CameraCaptureSession.StateCallback() {
                override fun onConfigured(session: CameraCaptureSession) {
                    captureSession = session
                    previewRequestBuilder.set(CaptureRequest.CONTROL_MODE, CameraMetadata.CONTROL_MODE_AUTO)
                    session.setRepeatingRequest(previewRequestBuilder.build(), null, null)
                    onPreviewStarted()
                }

                override fun onConfigureFailed(session: CameraCaptureSession) {
                    onPreviewFailed()
                }
            }, null)

        } catch (e: CameraAccessException) {
            Log.e(TAG, "startPreview: CameraAccessException ${e.message}")
            onPreviewFailed()
        }
    }

    // إعداد MediaRecorder للتسجيل
    private fun setupMediaRecorder(outputFilePath: String) {
        if (mediaRecorder == null) {
            mediaRecorder = MediaRecorder()
        } else {
            mediaRecorder!!.reset()
        }

        mediaRecorder!!.apply {
            setAudioSource(MediaRecorder.AudioSource.MIC)
            setVideoSource(MediaRecorder.VideoSource.SURFACE)
            setOutputFormat(MediaRecorder.OutputFormat.MPEG_4)
            setOutputFile(outputFilePath)
            setVideoEncodingBitRate(10000000)
            setVideoFrameRate(30)
            setVideoSize(1920, 1080)
            setVideoEncoder(MediaRecorder.VideoEncoder.H264)
            setAudioEncoder(MediaRecorder.AudioEncoder.AAC)
            prepare()
        }
    }

    // بدء تسجيل الفيديو
    fun startRecording(onRecordingStarted: () -> Unit, onError: (String) -> Unit) {
        if (isRecording) {
            onError("Already recording")
            return
        }

        val outputFilePath = getVideoFilePath()
        try {
            setupMediaRecorder(outputFilePath)
        } catch (e: Exception) {
            onError("MediaRecorder setup failed: ${e.message}")
            return
        }

        val camera = cameraDevice ?: run {
            onError("CameraDevice is null")
            return
        }

        try {
            val recorderSurface = mediaRecorder!!.surface
            val previewRequestBuilder = camera.createCaptureRequest(CameraDevice.TEMPLATE_RECORD)
            previewRequestBuilder.addTarget(previewSurface)
            previewRequestBuilder.addTarget(recorderSurface)

            camera.createCaptureSession(listOf(previewSurface, recorderSurface), object : CameraCaptureSession.StateCallback() {
                override fun onConfigured(session: CameraCaptureSession) {
                    captureSession = session
                    previewRequestBuilder.set(CaptureRequest.CONTROL_MODE, CameraMetadata.CONTROL_MODE_AUTO)
                    session.setRepeatingRequest(previewRequestBuilder.build(), null, null)
                    mediaRecorder!!.start()
                    isRecording = true
                    onRecordingStarted()
                }

                override fun onConfigureFailed(session: CameraCaptureSession) {
                    onError("CaptureSession configuration failed")
                }
            }, null)

        } catch (e: CameraAccessException) {
            onError("CameraAccessException: ${e.message}")
        }
    }

    // إيقاف التسجيل
    fun stopRecording(onStopped: () -> Unit, onError: (String) -> Unit) {
        if (!isRecording) {
            onError("Not currently recording")
            return
        }
        try {
            mediaRecorder?.stop()
            mediaRecorder?.reset()
            isRecording = false
            onStopped()
        } catch (e: Exception) {
            onError("Failed to stop recording: ${e.message}")
        }
    }

    // التبديل بين الكاميرا الأمامية والخلفية
    fun switchCamera(onSwitched: () -> Unit, onError: (String) -> Unit) {
        if (isRecording) {
            onError("Cannot switch camera during recording")
            return
        }
        isFrontCamera = !isFrontCamera
        closeCamera()
        openCamera({
            onSwitched()
        }, onError)
    }

    // إغلاق الكاميرا وتحرير الموارد
    fun closeCamera() {
        captureSession?.close()
        captureSession = null
        cameraDevice?.close()
        cameraDevice = null
        mediaRecorder?.release()
        mediaRecorder = null
        isRecording = false
    }

    // إنشاء مسار حفظ الفيديو
    private fun getVideoFilePath(): String {
        val dir = context.getExternalFilesDir(Environment.DIRECTORY_MOVIES)
        return "${dir?.absolutePath}/${SimpleDateFormat("yyyyMMdd_HHmmss", Locale.US).format(Date())}.mp4"
    }
}

2. تحديث الـ Activity لاستخدام CameraController

class CameraActivity : AppCompatActivity() {

    private lateinit var surfaceView: SurfaceView
    private lateinit var btnSwitchCamera: Button
    private lateinit var btnRecord: Button

    private lateinit var cameraManager: CameraManager
    private lateinit var cameraController: CameraController

    private val TAG = "CameraActivity"

    companion object {
        const val REQUEST_CODE_PERMISSIONS = 100
        val REQUIRED_PERMISSIONS = arrayOf(Manifest.permission.CAMERA, Manifest.permission.RECORD_AUDIO)
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_camera)

        surfaceView = findViewById(R.id.surfaceView)
        btnSwitchCamera = findViewById(R.id.btnSwitchCamera)
        btnRecord = findViewById(R.id.btnRecord)

        cameraManager = getSystemService(Context.CAMERA_SERVICE) as CameraManager

        surfaceView.holder.addCallback(object : SurfaceHolder.Callback {
            override fun surfaceCreated(holder: SurfaceHolder) {
                if (allPermissionsGranted()) {
                    initCameraController(holder.surface)
                } else {
                    ActivityCompat.requestPermissions(this@CameraActivity, REQUIRED_PERMISSIONS, REQUEST_CODE_PERMISSIONS)
                }
            }

            override fun surfaceChanged(holder: SurfaceHolder, format: Int, width: Int, height: Int) {}
            override fun surfaceDestroyed(holder: SurfaceHolder) {
                cameraController.closeCamera()
            }
        })

        btnSwitchCamera.setOnClickListener {
            cameraController.switchCamera({
                cameraController.startPreview({}, {})
            }, {
                Log.e(TAG, "Switch camera error: $it")
            })
        }

        btnRecord.setOnClickListener {
            if (cameraController.isRecording) {
                cameraController.stopRecording({
                    runOnUiThread {
                        btnRecord.text = "Start Recording"
                    }
                }, {
                    Log.e(TAG, "Stop recording error: $it")
                })
            } else {
                cameraController.startRecording({
                    runOnUiThread {
                        btnRecord.text = "Stop Recording"
                    }
                }, {
                    Log.e(TAG, "Start recording error: $it")
                })
            }
        }
    }

    private fun initCameraController(surface: Surface) {
        cameraController = CameraController(this, cameraManager, surface)
        cameraController.openCamera({
            cameraController.startPreview({}, {})
        }, {
            Log.e(TAG, "Open camera error: $it")
        })
    }

    private fun allPermissionsGranted() = REQUIRED_PERMISSIONS.all {
        ActivityCompat.checkSelfPermission(baseContext, it) == PackageManager.PERMISSION_GRANTED
    }

    override fun onRequestPermissionsResult(requestCode: Int, permissions: Array<String>, grantResults: IntArray) {
        if (requestCode == REQUEST_CODE_PERMISSIONS) {
            if (allPermissionsGranted()) {
                initCameraController(surfaceView.holder.surface)
            } else {
                // أذونات غير مكتملة: يمكنك إظهار رسالة أو إغلاق التطبيق
                finish()
            }
        }
    }

    override fun onPause() {
        super.onPause()
        cameraController.closeCamera()
    }
}

ملاحظات:

الكود منظم وفصل منطق الكاميرا عن الـ UI.

يمكن توسيع CameraController بسهولة بإضافة مزايا أخرى (مثل التقاط الصور، التقاط لقطات الشاشة، التحكم بالكاميرا أكثر).

معالجة الأخطاء تتم بإرسال رسائل مناسبة.

جميع العمليات التي قد تؤثر على الـ UI تتم عبر runOnUiThread.
